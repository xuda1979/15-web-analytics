export function bucketMinutes(startMs, endMs) {

  const first = Math.floor(startMs / step) * step;

  constructor({ windowMs = 60_000, limit = 60, maxEntries = 10_000 } = {}) {
    this.windowMs = windowMs;                                         
    this.limit = limit;                                               
    this.maxEntries = maxEntries;                                     
    this.counts = new Map();                                          
  }                                                                   
  keyFor(now = Date.now()) {                                          
    return Math.floor(now / this.windowMs) * this.windowMs;           
  }                                                                   
  tick(now = Date.now()) {                                            
    if (this.counts.size < this.maxEntries) return;                   
    const cutoff = this.keyFor(now) - this.windowMs;                  
    for (const [key, entry] of this.counts) {                         
      if (entry.winStart < cutoff) this.counts.delete(key);           
    }                                                                 
  }                                                                   
  check(key, now = Date.now()) {                                      
    this.tick(now);                                                   
    const k = this.keyFor(now);                                       
    const entry = this.counts.get(key);                               
    if (!entry || entry.winStart !== k) {                             
      this.counts.set(key, { winStart: k, count: 1 });                
      return 0;                                                       
    }                                                                 
    if (entry.count >= this.limit) {                                  
      return this.windowMs - (now % this.windowMs);                   
    }                                                                 
    entry.count += 1;                                                 
    return 0;                                                         
  }                                                                   
}

       visits = visits + 1,                                           
       uniques = uniques + excluded.uniques`,                         
  ).run(...Object.values(where), args.u);

export function toCsv(rows, columns) {                                
  const esc = (v) => {                                                
    const s = v === null || v === undefined ? '' : String(v);         
    return /[",\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;       
  };                                                                  
  const body = rows.map((r) => columns.map((c) => esc(r[c])).join(','));
  return [columns.join(','), ...body].join('\n') + '\n';              
}

function parseRange(query, defaultDays = 30) {                        
  const now = Date.now();                                             
  let days = Number.parseInt(query.days, 10);                         
  if (!Number.isFinite(days) || days < 1) days = defaultDays;         
  if (days > 3650) days = 3650;                                       
  let end = now;                                                      
  if (query.to) {                                                     
    const t = Date.parse(query.to);                                   
    if (Number.isFinite(t)) end = t;                                  
  }                                                                   
  let start = end - days * 24 * 3600 * 1000;                          
  if (query.from) {                                                   
    const f = Date.parse(query.from);                                 
    if (Number.isFinite(f)) start = f;                                
  }                                                                   
  return { start, end };                                              
}                                                                     
function filteredQuery(filters) {

  if (filters.device) {

  }                                                                   
  return { sql: parts.join(' AND '), args };                          
}

function main() {                                                     
  const [, , cmd, ...rest] = process.argv;                            
  ensureDataDir();                                                    
  const db = openDb();                                                
  try {                                                               
    if (cmd === 'seed') {                                             
      seed(db, { verbose: true });                                    
    } else if (cmd === 'collect') {                                   
      collectOnce(db, rest);                                          
    } else if (cmd === 'prune') {                                     
      const days = int(pattern(rest, /^--days=(\d+)/), config.retentionDays || 30);
      const res = pruneEvents(db, Date.now() - days * 24 * 3600 * 1000);
      console.log(`prune: deleted ${res.deleted} events older than ${days} days`);
    } else if (cmd === 'audit') {                                     
      const report = generateAudit(db);                               
      highlight(report);                                              
    } else {                                                          
      console.log('usage: node src/cli.js <seed|collect|prune|audit>');
    }                                                                 
  } finally {                                                         
    db.close();                                                       
  }                                                                   
}                                                                     
function collectOnce(db, args) {                                      
  const secret = loadOrCreateSecret(config.dataDir);                  
  const options = collectArgs(args);                                  
  const ip = options['ip'] || '127.0.0.1';                            
  const ua = options['ua'] || 'cli/1.0';                              
  const event = normalizeEvent(                                       
    {                                                                 
      path: options['path'] || '/',                                   
      referrer: options['referrer'] || '',                            
      title: options['title'] || '',                                  
      sw: int(pattern(args, /^--sw=(\d+)/), 0),                       
      sh: int(pattern(args, /^--sh=(\d+)/), 0),                       
      lang: options['lang'] || '',                                    
    },                                                                
    { ip, ua },                                                       
    secret,                                                           
  );                                                                  
  const { id, isUnique } = insertEvent(db, event);                    
  console.log(`collect: id=${id} path=${event.path} unique=${isUnique}`);
}

function collectArgs(args) {                                          
  const opts = {};                                                    
  for (const a of args) {                                             
    const m = a.match(/^--([a-z-]+)=(.*)$/);                          
    if (m) opts[m[1]] = m[2];                                         
  }                                                                   
  return opts;                                                        
}                                                                     
function pattern(args, re) {                                          
  for (const a of args) {                                             
    const m = a.match(re);                                            
    if (m) return m[1];                                               
  }                                                                   
  return null;                                                        
}                                                                     
function highlight(report) {                                          
  console.log('Privacy audit report:');                               
  for (const check of report.checks) {                                
    console.log(`  [${check.status.toUpperCase()}] ${check.label}`);  
  }                                                                   
  console.log(                                                        
    `events=${report.stats.events} real=${report.stats.real_events} guests=${report.stats.guests} ip_rows=${report.stats.ip_rows}`,
  );                                                                  
  console.log(`db=${report.stats.db_path} (${report.stats.bytes_readable})`);
  console.log('Use /api/audit?format=text for the full plain-text report.');
}
