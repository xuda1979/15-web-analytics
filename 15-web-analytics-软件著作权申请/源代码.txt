  return fallback;
}

    db

export function loadOrCreateSecret(dataDir) {
  const secretFile = path.join(dataDir, '.wa_secret');
  try {
    const existing = fs.readFileSync(secretFile, 'utf8').trim();
    if (existing) return existing;
  } catch {
  }
  const secret =
    config.hashSalt !== 'wa-local-salt'
      ? crypto.createHash('sha256').update(config.hashSalt).digest('hex')
      : crypto.randomBytes(32).toString('hex');
  fs.writeFileSync(secretFile, secret, { mode: 0o600 });
  return secret;
}

export function anonymizeIp(ip) {
  if (!ip) return '';
  if (ip.includes(':')) {
    const parts = ip.split(':');
    return `${parts.slice(0, 3).join(':')}:xxxx:xxxx:xxxx:xxxx`;
  }
  const parts = ip.split('.');
  if (parts.length === 4) return `${parts[0]}.${parts[1]}.${parts[2]}.x`;
  return '<redacted>';
}

import { UAParser } from 'ua-parser-js';
export const PARSER_CACHE_MAX = 1000;
export const parserCache = new Map();
export function parseUa(ua) {
  const key = ua || '';
  const hit = parserCache.get(key);
  if (hit) {
    parserCache.delete(key);
    parserCache.set(key, hit);
    return hit;
  }
  const parser = new UAParser(key);
  const res = parser.getResult();
  const out = {
    browser: res.browser?.name || 'Unknown',
    browserVersion: res.browser?.major || res.browser?.version || '',
    os: res.os?.name || 'Unknown',
    device: res.device?.type || 'desktop',
    deviceModel: res.device?.model || '',
  };
  parserCache.set(key, out);
  if (parserCache.size > PARSER_CACHE_MAX) {
    parserCache.delete(parserCache.keys().next().value);
  }
  return out;
}

export class RateLimiter {
  constructor({ windowMs = 60_000, limit = 60, maxEntries = 10_000 } = {}) {

    this.limit = limit;
    this.maxEntries = maxEntries;
    this.counts = new Map();
  }
  keyFor(now = Date.now()) {
    return Math.floor(now / this.windowMs) * this.windowMs;
  }
  tick(now = Date.now()) {
    if (this.counts.size < this.maxEntries) return;
    const cutoff = this.keyFor(now) - this.windowMs;
    for (const [key, entry] of this.counts) {
      if (entry.winStart < cutoff) this.counts.delete(key);
    }
  }
  _touch(key, entry) {
    this.counts.delete(key);
    this.counts.set(key, entry);
  }
  check(key, now = Date.now()) {
    this.tick(now);
    const k = this.keyFor(now);
    const entry = this.counts.get(key);
    if (!entry || entry.winStart !== k) {
      if (this.counts.size >= this.maxEntries) {
        this.counts.delete(this.counts.keys().next().value);
      }
      this.counts.set(key, { winStart: k, count: 1 });
      return 0;
    }
    const retry = entry.count >= this.limit ? this.windowMs - (now % this.windowMs) : 0;
    if (retry === 0) entry.count += 1;
    this._touch(key, entry);
    return retry;
  }
}
